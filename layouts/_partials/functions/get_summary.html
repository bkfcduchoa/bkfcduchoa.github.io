{{/* Get page summary with fallback priority: Params.summary > Params.description > Params.abstract > Summary */}} {{/* Local copy of the Kit's function. Hugo's automatic .Summary is now the HTML of the first blocks (a code block and an image for `sonar`), which leaked into lists, og:description and JSON-LD. The automatic summary is built here as the old Hugo did it: the page's plain text, cut after `summaryLength` words (30, see hugo.yaml) at the end of that sentence. */}} {{ $summary := .Params.summary | default .Params.description | default .Params.abstract }} {{ if and (not $summary) .Content }} {{ $limit := 30 }} {{ $out := slice }} {{ $done := false }} {{ range $i, $word := split (.Plain | replaceRE `\s+` " " | strings.TrimSpace) " " }} {{ if not $done }} {{ $out = $out | append $word }} {{ if and (ge (add $i 1) $limit) (findRE `[.!?]$` $word 1) }}{{ $done = true }}{{ end }} {{ end }} {{ end }} {{ $summary = delimit $out " " | safeHTML }}{{/* .Plain keeps HTML entities, so this is HTML */}} {{ end }} {{ return $summary }}
