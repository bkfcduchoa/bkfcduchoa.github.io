{{- /* Images: a page resource first, then assets/media/, else the URL as written. Raster images get resized webp versions; GIF (may be animated) and SVG stay as they are. */ -}} {{- $u := urls.Parse .Destination }} {{- $img := "" }} {{- if not $u.IsAbs }} {{- with .PageInner.Resources.Get $u.Path }}{{ $img = . }} {{- else }}{{ with resources.Get (printf "media/%s" (strings.TrimPrefix "/" $u.Path)) }}{{ $img = . }}{{ end }} {{- end }} {{- end }} {{- if and $img (in (slice "jpeg" "png" "webp" "bmp" "tiff") $img.MediaType.SubType) }} {{- /* Keep the original at its old URL; pages show the resized copies. */}} {{- $img.Publish }} {{- partial "responsive-image.html" (dict "img" $img "alt" .PlainText "title" .Title) }} {{- else }} {{- end }} {{- /**/ -}}
